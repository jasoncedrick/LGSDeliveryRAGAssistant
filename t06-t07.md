### Step-by-Step Execution of Task T06 (Evaluation Harness Loop)

Under Section 3.2.4 and Section 3.2.6 of your methodology, the evaluation harness must ingest normalized (or baseline) queries, query the BM25 and FAISS indexes, execute Min-Max score normalization, merge the results using linear weighted fusion, and compute ranking metrics: **Hit@1, Hit@3, Hit@5, Mean Reciprocal Rank (MRR), and Context Precision**.

$$\text{Score}_{\text{hybrid}}(d) = \alpha \cdot \bar{S}_{\text{BM25}}(d) + \beta \cdot \bar{S}_{\text{FAISS}}(d)$$

$$\text{where } \bar{S} = \frac{S - S_{\min}}{S_{\max} - S_{\min} + \epsilon}, \quad \alpha + \beta = 1.0$$

#### Step 1: Implement the Headless Evaluation Engine (`scripts/run_eval_harness.py`)

Create `scripts/run_eval_harness.py` to evaluate your 30 development groups across different fusion weights:

```python
import os
import json
import pickle
import faiss
import numpy as np
from typing import List, Dict, Any, Tuple
from sentence_transformers import SentenceTransformer

DEV_BENCHMARK_PATH = "data/benchmark_scenarios_dev.json"
BM25_INDEX_PATH = "data/bm25_index.pkl"
FAISS_INDEX_PATH = "data/faiss_index.bin"
EMBED_MODEL_NAME = "sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"

class HybridEvaluationHarness:
    def __init__(self):
        print("[*] Initializing Evaluation Harness...")
        # 1. Load BM25 Index
        with open(BM25_INDEX_PATH, "rb") as f:
            bm25_data = pickle.load(f)
            self.bm25 = bm25_data["bm25"]
            self.corpus = bm25_data["chunks"]
            self.chunk_id_to_idx = {c["chunk_id"]: idx for idx, c in enumerate(self.corpus)}

        # 2. Load FAISS Dense Index
        self.faiss_index = faiss.read_index(FAISS_INDEX_PATH)
        
        # 3. Load Embedding Model (CPU safe)
        self.embedder = SentenceTransformer(EMBED_MODEL_NAME, device="cpu")

        # 4. Load Benchmark Scenarios
        with open(DEV_BENCHMARK_PATH, "r", encoding="utf-8") as f:
            self.scenarios = json.load(f)
        print(f"    Loaded {len(self.corpus)} corpus chunks and {len(self.scenarios)} dev scenario groups.")

    def _min_max_scale(self, scores: np.ndarray) -> np.ndarray:
        s_min = np.min(scores)
        s_max = np.max(scores)
        if s_max == s_min:
            return np.zeros_like(scores)
        return (scores - s_min) / (s_max - s_min + 1e-9)

    def retrieve(self, query: str, alpha: float = 0.5, beta: float = 0.5, top_k: int = 5) -> List[Dict[str, Any]]:
        """
        Executes hybrid score normalization and fusion.
        alpha: weight for BM25 sparse score
        beta: weight for FAISS dense score (cosine / inner product)
        """
        # --- BM25 Scoring ---
        tokenized_query = query.lower().split()
        bm25_raw_scores = np.array(self.bm25.get_scores(tokenized_query))
        bm25_norm = self._min_max_scale(bm25_raw_scores)

        # --- FAISS Dense Scoring ---
        query_vec = self.embedder.encode([query], normalize_embeddings=True)
        # Search all documents for full corpus normalization
        dense_raw_scores, dense_indices = self.faiss_index.search(
            query_vec.astype(np.float32), k=len(self.corpus)
        )
        dense_scores_array = np.zeros(len(self.corpus))
        for score, idx in zip(dense_raw_scores[0], dense_indices[0]):
            dense_scores_array[idx] = score
        dense_norm = self._min_max_scale(dense_scores_array)

        # --- Linear Weighted Fusion ---
        hybrid_scores = (alpha * bm25_norm) + (beta * dense_norm)

        # Extract Top-K ranked indices
        top_indices = np.argsort(hybrid_scores)[::-1][:top_k]

        results = []
        for rank, idx in enumerate(top_indices, 1):
            results.append({
                "rank": rank,
                "chunk_id": self.corpus[idx]["chunk_id"],
                "score": float(hybrid_scores[idx]),
                "pillar": self.corpus[idx]["governance_pillar"]
            })
        return results

    def evaluate_retrieval(self, alpha: float = 0.5, beta: float = 0.5, top_k: int = 5) -> Dict[str, Any]:
        """
        Runs batch evaluation across all 30 groups and 3 language variants (90 queries).
        Calculates Hit@1, Hit@3, Hit@5, and MRR.
        """
        hits = {1: 0, 3: 0, 5: 0}
        reciprocal_ranks = []
        precision_scores = []
        total_queries = 0

        for sc in self.scenarios:
            gt_ids = set(sc["ground_truth_chunk_ids"])

            for lang, query_text in sc["queries"].items():
                total_queries += 1
                retrieved = self.retrieve(query_text, alpha=alpha, beta=beta, top_k=top_k)
                retrieved_ids = [r["chunk_id"] for r in retrieved]

                # Hit@K evaluation
                for k in [1, 3, 5]:
                    if any(cid in gt_ids for cid in retrieved_ids[:k]):
                        hits[k] += 1

                # MRR calculation
                rr = 0.0
                for rank, cid in enumerate(retrieved_ids, 1):
                    if cid in gt_ids:
                        rr = 1.0 / rank
                        break
                reciprocal_ranks.append(rr)

                # Context Precision at K
                relevant_count = 0
                precision_at_k = []
                for rank, cid in enumerate(retrieved_ids, 1):
                    if cid in gt_ids:
                        relevant_count += 1
                        precision_at_k.append(relevant_count / rank)
                precision_scores.append(np.mean(precision_at_k) if precision_at_k else 0.0)

        return {
            "alpha": alpha,
            "beta": beta,
            "total_queries": total_queries,
            "hit_at_1": round(hits[1] / total_queries, 4),
            "hit_at_3": round(hits[3] / total_queries, 4),
            "hit_at_5": round(hits[5] / total_queries, 4),
            "mrr": round(float(np.mean(reciprocal_ranks)), 4),
            "mean_context_precision": round(float(np.mean(precision_scores)), 4)
        }

if __name__ == "__main__":
    harness = HybridEvaluationHarness()
    # Baseline test run: Equal weight fusion
    print("\n[*] Running baseline hybrid evaluation (alpha=0.5, beta=0.5)...")
    metrics = harness.evaluate_retrieval(alpha=0.5, beta=0.5, top_k=5)
    print("\n================ RETRIEVAL BENCHMARK RESULTS ================")
    for k, v in metrics.items():
        print(f"  {k:25}: {v}")
    print("=============================================================")

```

### Step 3: Implement Task T07 Part 1 (Retrieval Fusion Grid Search)

Under Section 3.2.6, you must systematically search for the weights ($\alpha, \beta$) that maximize **Context Precision** on the development set before freezing the pipeline for the held-out evaluation.

Create `scripts/grid_search_weights.py`:

```python
import pandas as pd
from scripts.run_eval_harness import HybridEvaluationHarness

def execute_grid_search():
    harness = HybridEvaluationHarness()
    
    # Grid testing alpha from 0.0 to 1.0 with step 0.1 (beta = 1.0 - alpha)
    grid_points = [round(x * 0.1, 1) for x in range(11)]
    records = []

    print("[*] Starting Retrieval Weight Optimization Grid Search...")
    for alpha in grid_points:
        beta = round(1.0 - alpha, 1)
        print(f"    Evaluating Config: alpha={alpha:.1f} (BM25), beta={beta:.1f} (FAISS)...")
        res = harness.evaluate_retrieval(alpha=alpha, beta=beta, top_k=5)
        records.append(res)

    df = pd.DataFrame(records)
    df = df.sort_values(by="mean_context_precision", ascending=False)
    
    print("\n================ OPTIMIZATION GRID SEARCH RESULTS ================")
    print(df.to_string(index=False))
    print("==================================================================")
    
    best_config = df.iloc[0]
    print(f"\n[+] Optimal Weights Identified: alpha={best_config['alpha']}, beta={best_config['beta']}")
    print(f"    Peak Context Precision: {best_config['mean_context_precision']}")
    print(f"    Peak Hit@5: {best_config['hit_at_5']}")
    
    df.to_csv("data/retrieval_grid_search_results.csv", index=False)
    print("    [+] Exported grid search logs to: data/retrieval_grid_search_results.csv")

if __name__ == "__main__":
    execute_grid_search()

```
